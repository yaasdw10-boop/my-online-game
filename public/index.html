class GameScene extends Phaser.Scene { constructor() { super('GameScene'); } preload() { // ใช้ภาพพื้นหลังสเปซที่โหลดติดชัวร์ this.load.image('bg', 'https://labs.phaser.io/assets/skies/space3.png'); } create() { this.add.tileSprite(0, 0, 5000, 5000, 'bg').setAlpha(0.6); this.socket = io(); globalSocket = this.socket; this.otherPlayers = this.physics.add.group(); this.monstersGroup = this.physics.add.group(); this.moveDir = { x: 0, y: 0 }; this.cameras.main.setZoom(1.5); this.socket.on('currentPlayers', (players) => { Object.keys(players).forEach(id => { if (id === this.socket.id) { myPlayerData = players[id]; this.addPlayer(players[id]); updateUI(); } else { this.addOther(players[id]); } }); }); this.socket.on('currentMonsters', (monsters) => { Object.keys(monsters).forEach(id => this.addMonster(monsters[id])); }); this.socket.on('newPlayer', (p) => this.addOther(p)); this.socket.on('disconnectPlayer', (id) => { this.otherPlayers.getChildren().forEach(o => { if (o.playerId === id) { if(o.shadow) o.shadow.destroy(); o.destroy(); } }); }); this.socket.on('playerMoved', (p) => { this.otherPlayers.getChildren().forEach(o => { if (o.playerId === p.id) { o.setPosition(p.x, p.y); if (o.shadow) o.shadow.setPosition(p.x, p.y + 18); if (o.nameText) o.nameText.setPosition(p.x, p.y - 35); } }); }); this.socket.on('updateStatsData', (p) => { myPlayerData = p; updateUI(); }); this.socket.on('playerDataUpdated', (p) => { myPlayerData = p; updateUI(); }); this.socket.on('updateInventory', (inv) => { if (myPlayerData) myPlayerData.inventory = inv; updateInventoryUI(); }); this.socket.on('monsterDamaged', (data) => { playSound('hit'); this.monstersGroup.getChildren().forEach(m => { if (m.monsterId === data.monsterId) showFloatingText(m.x, m.y - 20, `-${data.damage}`, '#ef4444'); }); }); this.socket.on('monsterDead', (data) => { this.monstersGroup.getChildren().forEach(m => { if (m.monsterId === data.monsterId) { showFloatingText(m.x, m.y - 20, `+${data.exp} EXP`, '#34d399'); if (selectedTarget && selectedTarget.monsterId === data.monsterId) { clearTarget(); } m.destroy(); } }); if (data.killerId === this.socket.id) { myPlayerData = data.player; if (data.leveledUp) { playSound('levelup'); } updateUI(); } }); this.socket.on('monsterRespawn', (m) => this.addMonster(m)); this.socket.on('chatMessage', (data) => { let box = document.getElementById('chat-box'); box.innerHTML += `
${data.name}: ${data.msg}
`; box.scrollTop = box.scrollHeight; }); let joy = document.getElementById('joystick'), knob = document.getElementById('knob'); let active = false, maxD = 35; joy.onpointerdown = () => { active = true; }; window.onpointermove = (e) => { if (!active) return; let rect = joy.getBoundingClientRect(); let dx = e.clientX - (rect.left + rect.width / 2); let dy = e.clientY - (rect.top + rect.height / 2); let dist = Math.hypot(dx, dy); if (dist > maxD) { dx = (dx/dist)*maxD; dy = (dy/dist)*maxD; } knob.style.transform = `translate(${dx}px, ${dy}px)`; this.moveDir.x = dx / maxD; this.moveDir.y = dy / maxD; }; window.onpointerup = () => { active = false; knob.style.transform = `translate(0px, 0px)`; this.moveDir.x = 0; this.moveDir.y = 0; }; } addPlayer(p) { this.shipShadow = this.add.ellipse(p.x, p.y + 18, 20, 8, 0x000000, 0.4); // สร้างตัวละครด้วยวงรี/สี่เหลี่ยมพิกเซลจำลอง (Novice สีเหลืองทอง) แทนการโหลดรูปภาพภายนอก ป้องกันการเกิดกล่องเขียว this.ship = this.add.container(p.x, p.y); let body = this.add.rectangle(0, 0, 24, 36, 0xfbbf24).setStrokeStyle(2, 0xffffff); let head = this.add.circle(0, -20, 10, 0xffdead); this.ship.add([body, head]); this.physics.world.enable(this.ship); this.ship.body.setCollideWorldBounds(true); this.ship.body.setSize(24, 36); this.cameras.main.startFollow(this.ship, true, 0.1, 0.1); this.nameText = this.add.text(p.x, p.y - 42, `${p.name} (You)`, { fontSize: '10px', fill: '#fbbf24', stroke: '#000', strokeThickness: 3 }).setOrigin(0.5); } addOther(p) { let shadow = this.add.ellipse(p.x, p.y + 18, 20, 8, 0x000000, 0.4); let o = this.add.container(p.x, p.y); let body = this.add.rectangle(0, 0, 24, 36, 0x38bdf8).setStrokeStyle(2, 0xffffff); let head = this.add.circle(0, -20, 10, 0xffdead); o.add([body, head]); o.playerId = p.id; o.shadow = shadow; this.otherPlayers.add(o); o.nameText = this.add.text(p.x, p.y - 42, p.name, { fontSize: '9px', fill: '#fff', stroke: '#000', strokeThickness: 3 }).setOrigin(0.5); } addMonster(m) { // สร้างมอนสเตอร์ Poring สีชมพูกลมดุ๊กดิ๊ก let monsterSprite = this.add.container(m.x, m.y); let body = this.add.circle(0, 0, 14, 0xf472b6).setStrokeStyle(2, 0xffffff); monsterSprite.add(body); monsterSprite.monsterId = m.id; this.physics.world.enable(monsterSprite); monsterSprite.body.setCircle(14, -14, -14); monsterSprite.setInteractive(new Phaser.Geom.Circle(0, 0, 16), Phaser.Geom.Circle.Contains); monsterSprite.on('pointerdown', () => { selectedTarget = monsterSprite; document.getElementById('target-hud').style.display = 'block'; document.getElementById('target-name').innerText = `${m.name} [Lv.${m.level}]`; }); this.monstersGroup.add(monsterSprite); this.add.text(m.x, m.y - 22, `${m.name} [Lv.${m.level}]`, { fontSize: '9px', fill: '#f87171', stroke: '#000', strokeThickness: 2 }).setOrigin(0.5); } doAttack() { if (this.ship) { // เอฟเฟกต์กระโดดโจมตีเล็กน้อย this.tweens.add({ targets: this.ship, scaleX: 1.2, scaleY: 0.8, duration: 80, yoyo: true }); if (selectedTarget && selectedTarget.active) { let dist = Phaser.Math.Distance.Between(this.ship.x, this.ship.y, selectedTarget.x, selectedTarget.y); if (dist <= 120 && globalSocket) { globalSocket.emit('attackMonster', selectedTarget.monsterId); } else { showFloatingText(this.ship.x, this.ship.y - 50, 'เป้าหมายอยู่ไกลเกินไป!', '#fbbf24'); } } else { let nearestMonster = null; let minDist = 90; this.monstersGroup.getChildren().forEach(m => { let dist = Phaser.Math.Distance.Between(this.ship.x, this.ship.y, m.x, m.y); if (dist < minDist) { minDist = dist; nearestMonster = m; } }); if (nearestMonster && globalSocket) { globalSocket.emit('attackMonster', nearestMonster.monsterId); } } } } update() { if (this.ship && this.ship.body) { let speed = 180; this.ship.body.setVelocity(this.moveDir.x * speed, this.moveDir.y * speed); if (this.shipShadow) { this.shipShadow.setPosition(this.ship.x, this.ship.y + 18); } if (this.nameText) { this.nameText.setPosition(this.ship.x, this.ship.y - 42); } if (this.moveDir.x !== 0 || this.moveDir.y !== 0) { this.socket.emit('playerMovement', { x: this.ship.x, y: this.ship.y }); } } } }
