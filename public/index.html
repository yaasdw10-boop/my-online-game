addPlayer(p) { this.shipShadow = this.add.ellipse(p.x, p.y + 18, 22, 9, 0x000000, 0.4); // ดีไซน์ตัวละคร Novice คลาสสิก: ผมสีน้ำตาล, ผ้าผูกคอสีฟ้า, ชุดคลุมสีเขียว this.ship = this.add.container(p.x, p.y); let body = this.add.rectangle(0, 4, 22, 26, 0xa3e635).setStrokeStyle(2, 0x3f6212); // ชุดคลุมสีเขียว let collar = this.add.triangle(0, -6, -8, -4, 8, -4, 0, 2, 0x38bdf8); // ผ้าผูกคอสีฟ้า let head = this.add.circle(0, -16, 11, 0xffdead).setStrokeStyle(1, 0xd4b28c); // ใบหน้า let hair = this.add.ellipse(0, -22, 16, 10, 0x78350f); // ผมทรงโนวิซสีน้ำตาล let eyeLeft = this.add.circle(-3, -16, 1.5, 0x000000); let eyeRight = this.add.circle(3, -16, 1.5, 0x000000); this.ship.add([body, collar, head, hair, eyeLeft, eyeRight]); this.physics.world.enable(this.ship); this.ship.body.setCollideWorldBounds(true); this.ship.body.setSize(22, 32); this.cameras.main.startFollow(this.ship, true, 0.1, 0.1); this.nameText = this.add.text(p.x, p.y - 40, `${p.name} (You)`, { fontSize: '10px', fill: '#fbbf24', stroke: '#000', strokeThickness: 3 }).setOrigin(0.5); } addOther(p) { let shadow = this.add.ellipse(p.x, p.y + 18, 22, 9, 0x000000, 0.4); let o = this.add.container(p.x, p.y); let body = this.add.rectangle(0, 4, 22, 26, 0x38bdf8).setStrokeStyle(2, 0x0369a1); // ผู้เล่นอื่นใส่ชุดฟ้า let collar = this.add.triangle(0, -6, -8, -4, 8, -4, 0, 2, 0xfacc15); let head = this.add.circle(0, -16, 11, 0xffdead); let hair = this.add.ellipse(0, -22, 16, 10, 0x451a03); let eyeLeft = this.add.circle(-3, -16, 1.5, 0x000000); let eyeRight = this.add.circle(3, -16, 1.5, 0x000000); o.add([body, collar, head, hair, eyeLeft, eyeRight]); o.playerId = p.id; o.shadow = shadow; this.otherPlayers.add(o); o.nameText = this.add.text(p.x, p.y - 40, p.name, { fontSize: '9px', fill: '#fff', stroke: '#000', strokeThickness: 3 }).setOrigin(0.5); } updatePlayerCostume(spriteObj, eq) { if (eq && eq.head) { if (!spriteObj.hat) { // หมวกกระต่ายแฟชั่นวางทับบนผมพอดี spriteObj.hat = this.add.ellipse(0, -30, 18, 12, 0xf472b6).setStrokeStyle(1, 0xffffff); spriteObj.add(spriteObj.hat); } } else if (spriteObj.hat) { spriteObj.hat.destroy(); spriteObj.hat = null; } }
